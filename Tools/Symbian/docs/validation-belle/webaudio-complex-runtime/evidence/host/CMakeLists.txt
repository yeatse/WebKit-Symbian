cmake_minimum_required(VERSION 3.28)
project(LLVMComplexContract LANGUAGES C)
set(builtins "${CMAKE_CURRENT_SOURCE_DIR}/../source/compiler-rt/lib/builtins")
set(tests "${CMAKE_CURRENT_SOURCE_DIR}/../source/compiler-rt/test/builtins/Unit")
enable_testing()
foreach(kind IN ITEMS muldc3 divdc3)
  add_executable(${kind}_test "${builtins}/${kind}.c" "${tests}/${kind}_test.c")
  target_compile_features(${kind}_test PRIVATE c_std_11)
  target_include_directories(${kind}_test PRIVATE "${builtins}")
  target_compile_options(${kind}_test PRIVATE -O2 -ffp-contract=off -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(${kind}_test PRIVATE -fsanitize=address,undefined)
  target_link_libraries(${kind}_test PRIVATE m)
  add_test(NAME upstream_${kind} COMMAND ${kind}_test)
endforeach()
