#!/usr/bin/env python3
"""Local native-UI fixture and bounded multipart upload verifier."""

import argparse
from email import policy
from email.parser import BytesParser
import hashlib
from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer
import json
from pathlib import Path
from urllib.parse import unquote, urlsplit

ROOT = Path(__file__).resolve().parent
SAMPLE_NAME = "上传-你好.txt"
SAMPLE = (ROOT / SAMPLE_NAME).read_bytes()
MAX_BODY = 16384


def parse_upload(content_type, body):
    if not content_type.startswith("multipart/form-data;") or len(body) > MAX_BODY:
        raise ValueError("expected bounded multipart/form-data")
    message = BytesParser(policy=policy.default.clone(utf8=True)).parsebytes(
        b"Content-Type: " + content_type.encode("ascii") + b"\r\nMIME-Version: 1.0\r\n\r\n" + body)
    parts = list(message.iter_parts())
    if message.defects or len(parts) != 1 or parts[0].defects or parts[0].is_multipart():
        raise ValueError("malformed multipart body")
    if parts[0].get_content_disposition() != "form-data" or parts[0].get_param("name", header="content-disposition") != "file":
        raise ValueError("expected exactly one file field")
    part = parts[0]
    name = part.get_filename()
    payload = part.get_payload(decode=True)
    if not name or not isinstance(payload, bytes):
        raise ValueError("missing filename or body")
    return {"name": name, "bytes": len(payload), "sha256": hashlib.sha256(payload).hexdigest(),
            "contentMatches": name == SAMPLE_NAME and payload == SAMPLE}


class Handler(BaseHTTPRequestHandler):
    def respond(self, status, payload, content_type):
        self.send_response(status)
        self.send_header("Content-Type", content_type)
        self.send_header("Content-Length", str(len(payload)))
        self.send_header("Cache-Control", "no-store")
        self.end_headers()
        self.wfile.write(payload)

    def do_GET(self):
        name = unquote(urlsplit(self.path).path).lstrip("/") or "index.html"
        allowed = {"index.html": "text/html; charset=utf-8", "suite.js": "text/javascript; charset=utf-8",
                   SAMPLE_NAME: "text/plain; charset=utf-8"}
        if name not in allowed:
            self.respond(404, b"Not found", "text/plain")
            return
        self.respond(200, (ROOT / name).read_bytes(), allowed[name])

    def do_POST(self):
        try:
            if urlsplit(self.path).path != "/upload":
                raise ValueError("unknown upload endpoint")
            origin = self.headers.get("Origin")
            if origin and origin != "http://" + self.headers.get("Host", ""):
                raise ValueError("cross-origin upload")
            size = int(self.headers.get("Content-Length", "0"))
            if not 0 < size <= MAX_BODY:
                raise ValueError("upload too large or empty")
            self.connection.settimeout(10)
            body = self.rfile.read(size)
            if len(body) != size:
                raise ValueError("truncated upload")
            result = parse_upload(self.headers.get("Content-Type", ""), body)
        except (OSError, ValueError, UnicodeError) as error:
            self.respond(400, json.dumps({"error": str(error)}).encode(), "application/json")
            return
        with self.server.evidence.open("a", encoding="utf-8") as stream:
            stream.write(json.dumps(result, ensure_ascii=False) + "\n")
        self.respond(200, json.dumps(result, ensure_ascii=False).encode(), "application/json; charset=utf-8")


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--host", default="127.0.0.1")
    parser.add_argument("--port", type=int, default=8782)
    parser.add_argument("--evidence", type=Path, required=True)
    args = parser.parse_args()
    args.evidence.parent.mkdir(parents=True, exist_ok=True)
    server = ThreadingHTTPServer((args.host, args.port), Handler)
    server.evidence = args.evidence
    print("Serving Belle UI fixture on", server.server_address, flush=True)
    try:
        server.serve_forever()
    finally:
        server.server_close()


if __name__ == "__main__":
    main()
