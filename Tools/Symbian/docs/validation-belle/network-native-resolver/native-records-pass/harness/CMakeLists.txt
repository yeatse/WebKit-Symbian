cmake_minimum_required(VERSION 3.28)
project(native_curl_records LANGUAGES CXX ASM)
include(SymbianApp)
symbian_add_executable(dns_native_contract probe.cpp curl-symbian-resolver.cpp)
target_include_directories(dns_native_contract PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_features(dns_native_contract PRIVATE cxx_std_17)
target_compile_options(dns_native_contract PRIVATE -marm -fno-rtti -fno-exceptions)
target_link_libraries(dns_native_contract PRIVATE SymbianNativeAbi Symbian::EUser "${SYMBIAN_SDK_PREFIX}/proxies/esock/esock.dso" "${SYMBIAN_SDK_PREFIX}/proxies/insock/insock.dso" "${SYMBIAN_SDK_PREFIX}/proxies/libc/libc.dso")
symbian_publish_executable(dns_native_contract UID3 0xe0000913 CAPABILITIES 0x2000)
