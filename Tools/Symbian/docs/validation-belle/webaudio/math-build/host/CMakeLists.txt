cmake_minimum_required(VERSION 3.28)
project(WebKitSymbianLog2f LANGUAGES CXX)
find_package(GTest CONFIG REQUIRED)
add_executable(log2f_contract actual-log2f.cpp test.cpp)
target_compile_features(log2f_contract PRIVATE cxx_std_20)
target_compile_options(log2f_contract PRIVATE -O2 -fsanitize=address,undefined -fno-omit-frame-pointer)
target_link_options(log2f_contract PRIVATE -fsanitize=address,undefined)
target_link_libraries(log2f_contract PRIVATE GTest::gtest_main)
enable_testing()
add_test(NAME actual_log2f_contract COMMAND log2f_contract)
