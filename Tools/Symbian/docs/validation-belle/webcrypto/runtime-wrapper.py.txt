url+='?autorun=1&run='+uuid.uuid4().hex
report={'status':'FAIL','suite':args.suite,'browser_sha256':args.sha256,'emulator_sha256':sha(Path(os.environ['EKA2L1'])),'url':url,'expected_checks':count};p=None;descendants={}

  assert sha(data/'drives/e/sys/bin/browser.exe')==args.sha256
  if args.suite=='fonts':
   source=repo/'tests/web-fonts';target=data/'drives/c/Data/WebKit/web-fonts';shutil.copytree(source,target,dirs_exist_ok=True)

     p=subprocess.Popen([str(repo/'eka2l1/run-browser.sh'),str(args.browser),url],stdout=stream,stderr=subprocess.STDOUT,env=env,start_new_session=True,pass_fds=(lock.fileno(),));deadline=time.monotonic()+75

   finally:
