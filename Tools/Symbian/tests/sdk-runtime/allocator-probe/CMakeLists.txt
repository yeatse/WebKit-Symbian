cmake_minimum_required(VERSION 3.28)
project(symbian_allocator_probe LANGUAGES CXX ASM)
include(SymbianApp)
if(NOT EXISTS "${PROBE_RUNTIME}")
    message(FATAL_ERROR "PROBE_RUNTIME must name a scratch runtime archive")
endif()
set_target_properties(SymbianRuntime PROPERTIES IMPORTED_LOCATION "${PROBE_RUNTIME}")
symbian_add_executable(allocator_probe probe.cpp)
target_compile_definitions(allocator_probe PRIVATE PROBE_MODE="${PROBE_MODE}")
symbian_publish_executable(allocator_probe UID3 ${PROBE_UID})
