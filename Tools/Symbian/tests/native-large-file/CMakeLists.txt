cmake_minimum_required(VERSION 3.28)
project(symbian_large_file_probe LANGUAGES CXX ASM)
include(SymbianApp)
find_package(MbedTLS CONFIG REQUIRED PATHS "${SYMBIAN_SDK_PREFIX}/lib/${SYMBIAN_TARGET_ARCH}/mbedtls/lib/cmake/MbedTLS" NO_DEFAULT_PATH)
symbian_add_executable(large_file_probe probe.cpp)
target_compile_features(large_file_probe PRIVATE cxx_std_17)
target_compile_options(large_file_probe PRIVATE -marm)
target_link_libraries(large_file_probe PRIVATE MbedTLS::mbedcrypto Symbian::EUser "${SYMBIAN_SDK_PREFIX}/proxies/efsrv/efsrv.dso")
symbian_publish_executable(large_file_probe UID3 0xe0000906 CAPABILITIES 0)
